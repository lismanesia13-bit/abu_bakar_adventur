{q:"Siapa yang ditugaskan mengumpulkan ayat Al-Qur'an?",a:["Zaid bin Tsabit","Abu Lahab","Abu Sufyan","Muawiyah"],c:0,e:"Zaid bin Tsabit ditugaskan mengumpulkan ayat Al-Qur'an."},
{q:"Apa tindakan Abu Bakar terhadap orang yang menolak membayar zakat?",a:["Membiarkannya","Menghapus zakat","Tegas mempertahankan kewajiban zakat","Mengganti zakat"],c:2,e:"Abu Bakar tegas mempertahankan kewajiban zakat."},
{q:"Sifat Abu Bakar yang patut diteladani adalah ...",a:["Sombong dan boros","Jujur, setia, dan dermawan","Pemarah","Malas"],c:1,e:"Beliau dikenal jujur, setia, dermawan, dan rendah hati."},
{q:"Bagaimana keteladanan Abu Bakar dalam menggunakan hartanya?",a:["Untuk berfoya-foya","Untuk menyombongkan diri","Untuk membantu perjuangan Islam","Untuk merugikan orang lain"],c:2,e:"Abu Bakar menginfakkan hartanya demi perjuangan Islam."}
];

let mode="single",names=[],scores=[],round=0,turn=0;
let sequence=[],timer=null,seconds=30,locked=false,history=[];

let board=[];

try{board=JSON.parse(localStorage.getItem("abuBoard")||"[]")}catch(e){}

const $=id=>document.getElementById(id);

function show(id){
["home","setup","game","result"].forEach(x=>$(x).classList.toggle("hidden",x!==id));
}

function escapeHTML(s){
return String(s).replace(/[&<>"']/g,c=>({
"&":"&amp;","<":"&lt;",">":"&gt;",'"':"&quot;","'":"&#39;"
}[c]));
}

function playTone(f=600,d=.1,type="sine"){
if(!sound)return;
try{
audio=audio||new(window.AudioContext||window.webkitAudioContext)();
let o=audio.createOscillator(),g=audio.createGain();
o.type=type;
o.frequency.value=f;
g.gain.setValueAtTime(.06,audio.currentTime);
g.gain.exponentialRampToValueAtTime(.001,audio.currentTime+d);
o.connect(g);g.connect(audio.destination);
o.start();o.stop(audio.currentTime+d);
}catch(e){}
}

function toggleSound(){
sound=!sound;
$("soundBtn").textContent=sound?"🔊 EFEK ON":"🔇 EFEK OFF";
playTone(650);
}

function toggleMusic(){
music=!music;
$("musicBtn").textContent=music?"♫ MUSIK ON":"♫ MUSIK OFF";

if(music){
try{
audio=audio||new(window.AudioContext||window.webkitAudioContext)();
audio.resume();
let notes=[523,659,784,698,784,880,784,659,587,698,784,988];
let i=0;
function play(){
if(!music)return;
let o=audio.createOscillator(),g=audio.createGain();
o.type="triangle";
o.frequency.value=notes[i++%notes.length];
g.gain.setValueAtTime(.0001,audio.currentTime);
g.gain.exponentialRampToValueAtTime(.022,audio.currentTime+.04);
g.gain.exponentialRampToValueAtTime(.0001,audio.currentTime+.3);
o.connect(g);g.connect(audio.destination);
o.start();o.stop(audio.currentTime+.32);
musicTimer=setTimeout(play,350);
}
play();
}catch(e){music=false}
}else{
clearTimeout(musicTimer);
}
}

function chooseMode(m){
mode=m;
$("setupTitle").textContent=m==="single"?"MODE TUNGGAL":"MODE TANDING";
$("player2").classList.toggle("hidden",m!=="duel");
$("name1").value="";
$("name2").value="";
show("setup");
playTone(650);
}

function startGame(){
let a=$("name1").value.trim();
let b=$("name2").value.trim();

if(!a||(mode==="duel"&&!b)){
alert("Silakan isi nama semua pemain!");
return;
}

if(mode==="duel"&&a.toLowerCase()===b.toLowerCase()){
alert("Gunakan nama yang berbeda!");
return;
}

names=mode==="single"?[a]:[a,b];
scores=names.map(()=>0);
sequence=shuffle(questions.map((_,i)=>i));
round=0;
turn=0;
history=[];
show("game");
drawQuestion();
}

function shuffle(arr){
let a=[...arr];
for(let i=a.length-1;i>0;i--){
let j=Math.floor(Math.random()*(i+1));
[a[i],a[j]]=[a[j],a[i]];
}
return a;
}

function updateScores(){
$("scoreboard").innerHTML=names.map((n,i)=>`
<div class="player ${i===turn?"active":""}">
<div>${i===turn?"🎯 GILIRAN":"👤"} ${escapeHTML(n)}</div>
<strong>${scores[i]}</strong> <small>PTS</small>
</div>
`).join("");
}

function drawQuestion(){
locked=false;
seconds=30;
$("clock").textContent=30;
$("feedback").classList.add("hidden");
$("nextBtn").classList.add("hidden");

let q=questions[sequence[round]];
$("round").textContent="SOAL "+(round+1)+" / 10";
$("bar").style.width=(round+1)*10+"%";
$("question").textContent=q.q;

updateScores();

$("options").innerHTML=q.a.map((v,i)=>`
<button class="answer" data-i="${i}">
<strong>${"ABCD"[i]}.</strong> ${escapeHTML(v)}
</button>
`).join("");

document.querySelectorAll("#options button").forEach(btn=>{
btn.addEventListener("click",()=>{
answer(Number(btn.dataset.i));
});
});

clearInterval(timer);

timer=setInterval(()=>{
seconds--;
$("clock").textContent=seconds;
if(seconds<=0)answer(-1);
},1000);
}

function answer(i){
if(locked)return;
locked=true;
clearInterval(timer);

let q=questions[sequence[round]];
let correct=i===q.c;
let points=correct?100:0;

scores[turn]+=points;
history.push({
player:turn,
correct:correct,
question:q.q,
answer:q.a[q.c]
});

document.querySelectorAll("#options button").forEach((b,j)=>{
b.disabled=true;
if(j===q.c)b.classList.add("good");
if(j===i&&i!==q.c)b.classList.add("bad");
});

$("feedback").classList.remove("hidden");
$("feedback").style.background=correct?"#d8f8e7":"#ffe4eb";
$("feedback").style.color=correct?"#08623d":"#92304e";

$("feedback").innerHTML=`
<strong>${correct?"🌟 JAWABAN BENAR!":"💡 BELUM TEPAT!"}</strong>
<p>${escapeHTML(q.e)}</p>
<strong>+${points} POIN</strong>
`;

$("nextBtn").classList.remove("hidden");
$("nextBtn").textContent=
mode==="duel"&&turn===0?"GILIRAN PEMAIN BERIKUTNYA ➜":
round===9?"LIHAT HASIL 🏆":"SOAL BERIKUTNYA ➜";

updateScores();
playTone(correct?790:230,.18);
}

function nextQuestion(){
if(!locked)return;

if(mode==="duel"&&turn===0){
turn=1;
drawQuestion();

}

turn=0;

if(round===9){
finishGame();
return;
}

round++;
drawQuestion();
}

{min:0,icon:"🥉",name:"PEMULA"},
{min:400,icon:"🥈",name:"CENDEKIA"},
{min:600,icon:"🥇",name:"JUARA"},
{min:800,icon:"👑",name:"MASTER ILMU"}
];

$("trophy").textContent=mode==="duel"&&scores[0]===scores[1]?"🤝":"🏆";

$("resultTitle").textContent=
mode==="single"?"HEBAT, "+names[0].toUpperCase()+"!":
scores[0]===scores[1]?"PERTANDINGAN SERI!":
"JUARA: "+names[winner].toUpperCase()+"!";

$("resultText").textContent=
"Selamat! Kamu telah menyelesaikan 10 tantangan.";

<div class="player">
<strong>${escapeHTML(n)}</strong>
<h2>${scores[i]}</h2>

});
});

board=board.slice(0,10);

try{
localStorage.setItem("abuBoard",JSON.stringify(board));
}catch(e){}

playTone(880,.3);
}

}
